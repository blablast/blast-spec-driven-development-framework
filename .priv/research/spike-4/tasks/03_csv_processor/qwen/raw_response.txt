```python
import csv
import os
import re
from typing import Dict, List, Any


def process_csv(path: str, schema: dict) -> dict:
    result = {
        "valid_rows": [],
        "errors": [],
        "summary": {"total": 0, "valid": 0, "errors": 0}
    }
    
    if not os.path.exists(path):
        result["errors"].append({
            "row_num": 0,
            "field": "file",
            "issue": "File not found"
        })
        return result
    
    try:
        with open(path, 'r', newline='', encoding='utf-8') as csvfile:
            reader = csv.DictReader(csvfile)
            row_num = 0
            
            for row in reader:
                row_num += 1
                # Skip blank rows
                if not any(row.values()):
                    result["summary"]["total"] += 1
                    continue
                    
                valid_row = {}
                row_errors = []
                
                for field_name, field_config in schema.items():
                    value = row.get(field_name)
                    
                    # Check required fields
                    if field_config.get("required", False) and (value is None or value == ""):
                        row_errors.append({
                            "row_num": row_num,
                            "field": field_name,
                            "issue": "Required field missing"
                        })
                        continue
                    
                    # Skip validation for optional missing fields
                    if value is None or value == "":
                        if not field_config.get("required", False):
                            valid_row[field_name] = None
                        continue
                    
                    # Type validation and coercion
                    try:
                        if field_config["type"] == "str":
                            valid_row[field_name] = str(value)
                        elif field_config["type"] == "int":
                            valid_row[field_name] = int(value)
                        elif field_config["type"] == "float":
                            valid_row[field_name] = float(value)
                        else:
                            raise ValueError(f"Unsupported type: {field_config['type']}")
                    except (ValueError, TypeError):
                        row_errors.append({
                            "row_num": row_num,
                            "field": field_name,
                            "issue": f"Invalid {field_config['type']}"
                        })
                        continue
                    
                    # Additional validations
                    if field_config["type"] == "int":
                        if "min" in field_config and valid_row[field_name] < field_config["min"]:
                            row_errors.append({
                                "row_num": row_num,
                                "field": field_name,
                                "issue": f"Value less than minimum {field_config['min']}"
                            })
                        if "max" in field_config and valid_row[field_name] > field_config["max"]:
                            row_errors.append({
                                "row_num": row_num,
                                "field": field_name,
                                "issue": f"Value greater than maximum {field_config['max']}"
                            })
                    
                    if field_config["type"] == "str":
                        if "min_len" in field_config and len(value) < field_config["min_len"]:
                            row_errors.append({
                                "row_num": row_num,
                                "field": field_name,
                                "issue": f"String length less than minimum {field_config['min_len']}"
                            })
                        
                        if "regex" in field_config:
                            if not re.match(field_config["regex"], value):
                                row_errors.append({
                                    "row_num": row_num,
                                    "field": field_name,
                                    "issue": "String does not match regex pattern"
                                })
                
                # Add to results
                result["summary"]["total"] += 1
                if row_errors:
                    result["errors"].extend(row_errors)
                    result["summary"]["errors"] += len(row_errors)
                else:
                    result["valid_rows"].append(valid_row)
                    result["summary"]["valid"] += 1
                    
    except Exception as e:
        result["errors"].append({
            "row_num": 0,
            "field": "file",
            "issue": str(e)
        })
    
    return result
```
