```python
import asyncio
from typing import Callable, Awaitable, TypeVar, Generic

T = TypeVar("T")


class AsyncWorkerPool(Generic[T]):
    def __init__(self, concurrency: int):
        self._semaphore = asyncio.Semaphore(concurrency)
        self._shutdown = False
        self._tasks: set[asyncio.Task] = set()

    async def submit(self, coro_factory: Callable[[], Awaitable[T]]) -> T:
        if self._shutdown:
            raise RuntimeError("Pool is shut down")
        async with self._semaphore:
            if self._shutdown:
                raise RuntimeError("Pool is shut down")
            task = asyncio.ensure_future(coro_factory())
            self._tasks.add(task)
            try:
                return await task
            finally:
                self._tasks.discard(task)

    async def map(self, coro_factory: Callable[[int], Awaitable[T]], items: list) -> list[T]:
        results = await asyncio.gather(*[self.submit(lambda i=item: coro_factory(i)) for item in items])
        return list(results)

    async def shutdown(self) -> None:
        self._shutdown = True
        if self._tasks:
            await asyncio.gather(*self._tasks, return_exceptions=True)
```
